async getEvent(
    accessToken: string,
    calendarId: string,
    eventId: string
  ): Promise<CalendarEvent> {
    // If config is already set, use it; otherwise parse from accessToken
    if (!this.config) {
      const { username, password, serverUrl } = this.parseCalDAVCredentials(accessToken);
      this.setConfig({
        serverUrl,
        username,
        password,
        calendarPath: calendarId,
      });
    } else {
      this.config.calendarPath = calendarId;
    }
    
    // First, try searching in the specified calendar
    let events = await this.searchEvents(accessToken, {
      calendarId,
      timeMin: new Date(0), // Search from epoch to get all events
      timeMax: new Date('2100-01-01'), // Far future date
      maxResults: 1000, // Large limit to find the event
    });
    
    // Find event by UID (eventId)
    let event = events.find(e => e.id === eventId);
    
    // If not found and calendarId was specified, try searching all calendars
    // This handles cases where the calendarId doesn't exactly match the calendar path
    if (!event && calendarId && calendarId !== 'primary') {
      logger.debug('Event not found in specified calendar, searching all calendars', {
        calendarId,
        eventId,
        service: 'caldavProvider',
      });
      events = await this.searchEvents(accessToken, {
        // Don't specify calendarId to search all calendars
        timeMin: new Date(0),
        timeMax: new Date('2100-01-01'),
        maxResults: 1000,
      });
      event = events.find(e => e.id === eventId);
    }
    
    if (!event) {
      logger.error('Event not found after searching', undefined, {
        calendarId,
        eventId,
        searchedEventsCount: events.length,
        availableEventIds: events.slice(0, 5).map(e => e.id),
        service: 'caldavProvider',
      });
      throw new Error('Event not found');
    }
    
    return event;
  }
